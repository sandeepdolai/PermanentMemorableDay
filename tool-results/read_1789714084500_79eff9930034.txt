    56→const BLOCK_BY_TYPE = Object.fromEntries(BLOCKS.map((b) => [b.type, b]));

    58→interface Block {

    63→interface Scene {
    64→  id: string;

    67→
    68→const MAX_SCENES = 8;
    69→
    70→/** Deterministic draft layouts — abstract, no themed content */
    71→const DRAFT_PATTERNS: string[][] = [
    72→  ["text", "photo"],
    73→  ["video", "text"],
    74→  ["gift", "confetti"],
    75→  ["countdown", "cta"],
    76→  ["quiz", "text"],
    77→  ["audio", "photo"],
    78→  ["text", "cta"],
    79→  ["reward", "confetti"],

    81→
    82→function seedScenes(opts: BuilderOptions): Scene[] {
    83→  if (opts.ai) {
    84→    return [
    85→      { id: "s1", blocks: [{ id: "b1", type: "text" }, { id: "b2", type: "photo" }] },
    86→      { id: "s2", blocks: [{ id: "b3", type: "gift" }, { id: "b4", type: "confetti" }] },
    87→      { id: "s3", blocks: [{ id: "b5", type: "text" }, { id: "b6", type: "cta" }] },
    88→    ];
    89→  }
    90→  const count = Math.max(1, Math.min(opts.scenes ?? 1, MAX_SCENES));
    91→  const scenes: Scene[] = [];
    92→  for (let i = 0; i < count; i++) {
    93→    const pattern = DRAFT_PATTERNS[(i + (opts.cover ?? 0)) % DRAFT_PATTERNS.length];
    94→    scenes.push({
    95→      id: `s${i + 1}`,
    96→      blocks: pattern.map((t, j) => ({ id: `b${i * 4 + j + 1}`, type: t })),
    97→    });
    98→  }
    99→  if (opts.initialBlock && BLOCK_BY_TYPE[opts.initialBlock]) {
   100→    scenes[0].blocks.push({ id: "bSeed", type: opts.initialBlock });
   101→  }
   102→  return scenes;
   103→}

   257→
   258→export function ExperienceBuilder({ opts, onClose }: { opts: BuilderOptions; onClose: () => void }) {
   259→  const { notify, openMoment, openShare, sheet, player } = useMD();
   260→  const [title, setTitle] = useState(opts.title ?? "Untitled Experience");
   261→  const [editing, setEditing] = useState(false);
   262→  const [scenes, setScenes] = useState<Scene[]>(() => seedScenes(opts));
   263→  const [sceneIdx, setSceneIdx] = useState(0);
   264→  const [selectedBlock, setSelectedBlock] = useState<string | null>(null);
   265→  const idRef = useRef(100);
   266→  const blocksEndRef = useRef<HTMLDivElement>(null);
   267→  const cover = opts.cover ?? 5;
   268→
   269→  // Escape closes the builder — deferred while a sheet or the player is layered above
   270→  useEffect(() => {
   271→    const onKey = (e: KeyboardEvent) => {
   272→      if (e.key === "Escape" && !sheet && !player) onClose();
   273→    };
   274→    window.addEventListener("keydown", onKey);
   275→    return () => window.removeEventListener("keydown", onKey);
   276→  }, [sheet, player, onClose]);
   277→
   278→  const scene = scenes[Math.min(sceneIdx, scenes.length - 1)];
   279→  const scenePos = sceneIdx + 1;
   280→
   281→  const addBlock = (type: string) => {
   282→    const id = `b${idRef.current++}`;
   283→    setScenes((prev) =>
   284→      prev.map((s, i) =>
   285→        i === sceneIdx ? { ...s, blocks: [...s.blocks, { id, type }] } : s
   286→      )
   287→    );
   288→    setSelectedBlock(id);
   289→    notify(`${BLOCK_BY_TYPE[type]?.label ?? "Block"} added to Scene ${scenePos}`);
   290→    requestAnimationFrame(() => {
   291→      blocksEndRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   292→    });
   293→  };
   294→
   295→  const removeBlock = (id: string) => {
   296→    setScenes((prev) =>
   297→      prev.map((s, i) => (i === sceneIdx ? { ...s, blocks: s.blocks.filter((b) => b.id !== id) } : s))
   298→    );
   299→    if (selectedBlock === id) setSelectedBlock(null);
   300→    notify("Block removed");
   301→  };
   302→
   303→  const addScene = () => {
   304→    if (scenes.length >= MAX_SCENES) {
   305→      notify(`Scene limit reached (${MAX_SCENES}) in this preview`);
   306→      return;
   307→    }
   308→    const id = `s${Date.now()}`;
   309→    setScenes((prev) => [...prev, { id, blocks: [] }]);
   310→    setSceneIdx(scenes.length);
   311→    setSelectedBlock(null);
   312→    notify(`Scene ${scenes.length + 1} added`);
   313→  };
   314→
   315→  const commitTitle = () => {
   316→    setEditing(false);
   317→    const t = title.trim();
   318→    if (t && t !== (opts.title ?? "Untitled Experience")) notify("Title updated");
   319→    if (!t) setTitle(opts.title ?? "Untitled Experience");
   320→  };
   321→
   322→  return (
   323→    <motion.div
   324→      initial={{ x: "100%" }}
   325→      animate={{ x: 0 }}
   326→      exit={{ x: "100%" }}
   327→      transition={{ type: "spring", stiffness: 380, damping: 40 }}
   328→      className="absolute inset-0 z-[60] flex flex-col bg-[#F5F5F7]"
   329→      role="dialog"
   330→      aria-modal="true"
   331→      aria-label="Experience builder"
   332→    >
   333→      {/* ---------- Top chrome ---------- */}
   334→      <header className="relative z-10 flex items-center gap-3 border-b border-[#1D1D1F]/[0.07] bg-white/85 px-4 pb-3 pt-4 shadow-[0_10px_30px_-18px_rgba(29,29,31,0.25)] backdrop-blur-xl">
   335→        <button
   336→          type="button"
   337→          onClick={onClose}
   338→          aria-label="Back to app"
   339→          className="flex h-[38px] w-[38px] shrink-0 items-center justify-center rounded-full bg-[#007AFF]/[0.1] text-[#007AFF] transition-transform active:scale-90"
   340→        >
   341→          <ChevronLeft size={20} strokeWidth={2.4} aria-hidden />
   342→        </button>
   343→
   344→        <div className="min-w-0 flex-1 text-center">
   345→          {editing ? (
   346→            <input
   347→              autoFocus
   348→              value={title}
   349→              maxLength={40}
   350→              onChange={(e) => setTitle(e.target.value)}
   351→              onBlur={commitTitle}
   352→              onKeyDown={(e) => {
   353→                if (e.key === "Enter") commitTitle();
   354→                if (e.key === "Escape") {
   355→                  setTitle(opts.title ?? "Untitled Experience");
   356→                  setEditing(false);
   357→                }
   358→              }}
   359→              aria-label="Experience title"
   360→              className="w-full rounded-[10px] bg-[#F5F5F7] px-2.5 py-1 text-center text-[15.5px] font-bold tracking-[-0.02em] text-[#1D1D1F] outline-none ring-2 ring-[#007AFF]"
   361→            />
   362→          ) : (
   363→            <button
   364→              type="button"
   365→              onClick={() => setEditing(true)}
   366→              aria-label="Rename experience"
   367→              className="max-w-full truncate rounded-[10px] px-2 py-0.5 text-[15.5px] font-bold tracking-[-0.02em] text-[#1D1D1F] transition-colors hover:bg-[#1D1D1F]/[0.04]"
   368→            >
   369→              {title}
   370→            </button>
   371→          )}
   372→          <p className="mt-0.5 flex items-center justify-center gap-1.5 text-[11px] font-medium text-[#AAAAAA]">
   373→            <Check size={10} strokeWidth={3} className="text-[#30D158]" aria-hidden /> Saved
   374→            {opts.ai ? (
   375→              <>
   376→                <span aria-hidden>·</span>
   377→                <Sparkles size={10} className="text-[#5E5CE6]" aria-hidden />
   378→                <span className="font-semibold text-[#5E5CE6]">AI sketch</span>
   379→              </>
   380→            ) : null}
   381→          </p>
   382→        </div>
   383→
   384→        <div className="flex shrink-0 items-center gap-2">
   385→          <button
   386→            type="button"
   387→            onClick={() =>
   388→              openMoment({
   389→                id: `preview-${Date.now()}`,
   390→                title: title || "Untitled Experience",
   391→                cover,
   392→                dedication: "Draft preview",
   393→              })
   394→            }
   395→            aria-label="Preview experience"
   396→            className="flex h-[38px] w-[38px] items-center justify-center rounded-full bg-[#007AFF] text-white pill-shadow transition-transform active:scale-90"
   397→          >
   398→            <Play size={16} fill="currentColor" aria-hidden />
   399→          </button>
   400→          <button
   401→            type="button"
   402→            onClick={() => openShare({ id: `draft-${opts.cover ?? 0}`, title: title || "Untitled Experience" })}
   403→            aria-label="Share draft"
   404→            className="flex h-[38px] w-[38px] items-center justify-center rounded-full bg-[#1D1D1F]/[0.06] text-[#1D1D1F] transition-transform active:scale-90"
   405→          >
   406→            <Share2 size={15.5} strokeWidth={2.2} aria-hidden />
   407→          </button>
   408→        </div>
   409→      </header>
   410→
   411→      {/* ---------- Scrollable workspace ---------- */}
   412→      <main className="no-scrollbar relative flex-1 overflow-y-auto overscroll-contain">
   413→        <div className="space-y-5 px-5 pb-6 pt-5">
   414→          {/* Scene storyboard */}
   415→          <section aria-label="Scenes">
   416→            <div className="mb-2.5 flex items-center justify-between px-0.5">
   417→              <h2 className="text-[13px] font-bold uppercase tracking-[0.07em] text-[#AAAAAA]">
   418→                Scenes · {scenes.length}
   419→              </h2>
   420→              <p className="text-[11.5px] font-medium text-[#AAAAAA]">Tap a scene to edit it</p>
   421→            </div>
   422→            <div className="-mx-5 flex gap-3 overflow-x-auto px-5 pb-1 no-scrollbar">
   423→              {scenes.map((s, i) => {
   424→                const active = i === sceneIdx;
   425→                return (
   426→                  <button
   427→                    key={s.id}
   428→                    type="button"
   429→                    onClick={() => {
   430→                      setSceneIdx(i);
   431→                      setSelectedBlock(null);
   432→                    }}
   433→                    aria-pressed={active}
   434→                    aria-label={`Scene ${i + 1}, ${s.blocks.length} blocks`}
   435→                    className={cn(
   436→                      "relative w-[104px] shrink-0 overflow-hidden rounded-[18px] bg-white text-left transition-all active:scale-[0.96]",
   437→                      active
   438→                        ? "card-shadow ring-2 ring-[#007AFF] ring-offset-2 ring-offset-[#F5F5F7]"
   439→                        : "card-shadow hairline"
   440→                    )}
   441→                  >
   442→                    <CoverArt variant={(cover + i) % 10} className="h-[64px] w-full">
   443→                      <span
   444→                        className={cn(
   445→                          "absolute left-2 top-2 flex h-[22px] w-[22px] items-center justify-center rounded-full text-[11px] font-bold backdrop-blur-md",
   446→                          active ? "bg-[#007AFF] text-white" : "bg-[#1D1D1F]/35 text-white"
   447→                        )}
   448→                      >
   449→                        {i + 1}
   450→                      </span>
   451→                    </CoverArt>
   452→                    <div className="px-2.5 py-2">
   453→                      <p className="text-[12px] font-bold tracking-[-0.01em] text-[#1D1D1F]">
   454→                        Scene {i + 1}
   455→                      </p>
   456→                      <p className="mt-0.5 text-[10.5px] font-medium text-[#AAAAAA]">
   457→                        {s.blocks.length} {s.blocks.length === 1 ? "block" : "blocks"}
   458→                      </p>
   459→                    </div>
   460→                  </button>
   461→                );
   462→              })}
   463→              <button
   464→                type="button"
   465→                onClick={addScene}
   466→                aria-label="Add scene"
   467→                className="flex w-[104px] shrink-0 flex-col items-center justify-center gap-2 rounded-[18px] border-2 border-dashed border-[#1D1D1F]/[0.15] bg-white/50 text-[#AAAAAA] transition-all hover:border-[#007AFF]/40 hover:text-[#007AFF] active:scale-[0.96]"
   468→              >
   469→                <span className="flex h-9 w-9 items-center justify-center rounded-full bg-[#1D1D1F]/[0.05]">
   470→                  <Plus size={18} strokeWidth={2.4} aria-hidden />
   471→                </span>
   472→                <span className="text-[11.5px] font-semibold">Scene</span>
   473→              </button>
   474→            </div>
   475→          </section>
   476→
   477→          {/* Scene canvas */}
   478→          <section aria-label={`Scene ${scenePos} canvas`} className="card-shadow hairline rounded-[24px] bg-white p-4">
   479→            <div className="mb-3 flex items-center justify-between">
