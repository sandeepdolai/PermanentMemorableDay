    47→/** iOS-style dark glass toast — optional action button (e.g. Undo) */
    48→function GlassToast({ message, action, onAction }: { message: string; action?: ToastAction; onAction: () => void }) {
    49→  return (
    50→    <div className="pointer-events-none absolute inset-x-0 top-[70px] z-[90] flex justify-center px-6">
    51→      <motion.div
    52→        initial={{ opacity: 0, y: -14, scale: 0.96 }}
    53→        animate={{ opacity: 1, y: 0, scale: 1 }}
    54→        exit={{ opacity: 0, y: -14, scale: 0.96 }}
    55→        transition={{ type: "spring", stiffness: 420, damping: 30 }}
    56→        className="pointer-events-auto relative flex max-w-full items-center gap-2 overflow-hidden rounded-full bg-[#1D1D1F]/88 px-4 py-2.5 shadow-[0_16px_40px_-10px_rgba(29,29,31,0.5)] md:max-w-[520px]"
    57→        style={{ WebkitBackdropFilter: "blur(20px)", backdropFilter: "blur(20px)" }}
    58→      >
    59→        <Info size={14} className="shrink-0 text-[#64D2FF]" aria-hidden />
    60→        <span className="truncate text-[13px] font-medium text-white">{message}</span>
    61→        {action ? (
    62→          <>
    63→            <span aria-hidden className="h-3.5 w-px shrink-0 bg-white/20" />
    64→            <button
    65→              type="button"
    66→              onClick={() => {
    67→                onAction();
    68→                action.onClick();
    69→              }}
    70→              className="shrink-0 rounded-full bg-white/[0.14] px-3 py-1 text-[12px] font-bold tracking-[-0.01em] text-white transition-colors hover:bg-white/25 active:scale-95"
    71→            >
    72→              {action.label}
    73→            </button>
    74→          </>
    75→        ) : null}
    76→        {/* auto-dismiss progress hairline — longer when an action needs a click */}
    77→        <motion.span
    78→          aria-hidden
    79→          initial={{ scaleX: 1 }}
    80→          animate={{ scaleX: 0 }}
    81→          transition={{ duration: action ? 4.2 : 2.4, ease: "linear" }}
    82→          className="absolute inset-x-0 bottom-0 h-[2.5px] origin-left rounded-full bg-[#64D2FF]/60"
    83→        />
    84→      </motion.div>
    85→    </div>
    86→  );
    87→}

   267→        if (!window.localStorage.getItem("md-onboarded")) setTourOpen(true);

   278→          setArchivedIds(JSON.parse(storedArchived) as string[]);
   279→        }
   280→      } catch {
   281→        // storage unavailable — skip the tour

   289→    const mql = window.matchMedia("(prefers-color-scheme: dark)");

   295→    if (theme === "system") {

   316→        window.localStorage.setItem("md-drafts", JSON.stringify(next));
   317→      } catch {
   318→        // ignore
   319→      }
   320→      return next;
   321→    });

   333→        // ignore

   534→  }, []);
   535→
   536→  const isSearchTab = tab === "home" || tab === "explore";
   537→
   538→  /** Derived live from the feed — badge + sidebar stay in sync automatically */
   539→  const unreadCount = useMemo(
   540→    () => notifications.filter((n) => n.unread).length,
   541→    [notifications]
   542→  );
   543→

   545→    <MotionConfig reducedMotion="user">
   546→      <MDContext.Provider
   547→        value={{
   548→          tab,
   549→          setTab,
   550→          query,
   551→          setQuery,
   552→          submitSearch,
   553→          notify,
   554→          openSheet,
   555→          sheet,
   556→          openMoment,
   557→          player,
   558→          closePlayer: closeMoment,
   559→          openShare,
   560→          unreadCount,
   561→          markAllRead,
   562→          notifications,
   563→          dismissNotification,
   564→          insertNotification,
   565→          markNotificationRead,
   566→          openBuilder,
   567→          builder,
   568→          closeBuilder,
   569→          openSettings,
   570→          settingsTopic,
   571→          openExplore,
   572→          exploreItem,
   573→          openInsights,
   574→          openComposer,
   575→          aiInsertRef,
   576→          insertAiMessage,
   577→          openAuth,
   578→          tourOpen,
   579→          openTour,

   581→          savedIds,
   582→          toggleSaved,

   584→          setTheme,
   585→          drafts,

   590→          archivedIds,
   591→          toggleArchived,
   592→          openStats,
   593→          paletteOpen,

   595→          closePalette,
   596→        }}
   597→      >
   598→      {/*

   625→              : "bg-gradient-to-b from-[#F5F5F7] via-[#F5F5F7]/80 to-transparent dark:from-[#0A0A0C] dark:via-[#0A0A0C]/80"
   626→          )}
   627→        >
   628→          {isSearchTab ? (
   629→            <div className="relative flex items-center gap-2.5">
   630→              <div className="min-w-0 flex-1 md:mx-auto md:max-w-[560px]">
   631→                <SearchBar />
