    42→/** One-shot confetti burst (gift reveal / confetti blocks) */
    43→function ConfettiBurst() {
    44→  const [particles] = useState<Particle[]>(() =>
    45→    Array.from({ length: 28 }, (_, i) => {
    46→      const angle = (i / 28) * 360 + (Math.random() - 0.5) * 18;
    47→      const distance = 90 + Math.random() * 120;
    48→      const rad = (angle * Math.PI) / 180;
    49→      return {
    50→        id: i,
    51→        x: Math.cos(rad) * distance,
    52→        y: Math.sin(rad) * distance,
    53→        size: 6 + Math.random() * 7,
    54→        rotate: Math.random() * 720 - 360,
    55→        delay: Math.random() * 0.14,
    56→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
    57→      };
    58→    })
    59→  );
    60→
    61→  return (
    62→    <div aria-hidden className="pointer-events-none absolute inset-0 flex items-center justify-center">
    63→      {particles.map((p) => (
    64→        <motion.span
    65→          key={p.id}
    66→          initial={{ x: 0, y: 0, opacity: 1, rotate: 0, scale: 0.5 }}
    67→          animate={{ x: p.x, y: p.y, opacity: 0, rotate: p.rotate, scale: 1 }}
    68→          transition={{ duration: 1.05, ease: "easeOut", delay: p.delay }}
    69→          className="absolute rounded-[2px]"
    70→          style={{ width: p.size, height: p.size * 0.72, backgroundColor: p.color }}
    71→        />
    72→      ))}
    73→    </div>
    74→  );
    75→}

   195→type SoundtrackState = "idle" | "playing" | "paused" | "needsTap";
   196→
   197→/** Plays the picked snippet on loop; falls back to a tap-to-start chip
   198→ *  when the browser blocks programmatic audio. */
   199→function useSoundtrack(music: SongPick | null) {
   200→  const [state, setState] = useState<SoundtrackState>("idle");
   201→  const audioRef = useRef<HTMLAudioElement | null>(null);
   202→  const url = music?.previewUrl ?? null;
   203→  const start = music?.start ?? 0;
   204→  const length = music?.length ?? 30;
   205→
   206→  useEffect(() => {
   207→    if (!url) return;
   208→    const audio = new Audio(url);
   209→    audio.preload = "auto";
   210→    audioRef.current = audio;
   211→
   212→    const begin = () => {
   213→      try {
   214→        audio.currentTime = start;
   215→      } catch {
   216→        // seek before metadata — retried on timeupdate anyway
   217→      }
   218→      audio
   219→        .play()
   220→        .then(() => setState("playing"))
   221→        .catch(() => setState("needsTap"));
   222→    };
   223→    const onTime = () => {
   224→      if (audio.currentTime >= start + length - 0.05) {
   225→        audio.currentTime = start; // loop the snippet
   226→      }
   227→    };
   228→    const onMeta = () => {
   229→      try {
   230→        audio.currentTime = start;
   231→      } catch {
   232→        // ignore
   233→      }
   234→    };
   235→    audio.addEventListener("timeupdate", onTime);
   236→    audio.addEventListener("loadedmetadata", onMeta, { once: true });
   237→    begin();
   238→    if (audio.readyState >= 1) {
   239→      try {
   240→        audio.currentTime = start;
   241→      } catch {
   242→        // ignore
   243→      }
   244→    }
   245→
   246→    return () => {
   247→      audio.removeEventListener("timeupdate", onTime);
   248→      audio.removeEventListener("loadedmetadata", onMeta);
   249→      audio.pause();
   250→      audioRef.current = null;
   251→    };
   252→  }, [url, start, length]);
   253→
   254→  const toggle = useCallback(() => {
   255→    const audio = audioRef.current;
   256→    if (!audio) return;
   257→    if (audio.paused) {
   258→      try {
   259→        audio.currentTime = start;
   260→      } catch {
   261→        // ignore
   262→      }
   263→      void audio.play().then(() => setState("playing")).catch(() => setState("needsTap"));
   264→    } else {
   265→      audio.pause();
   266→      setState("paused");
   267→    }
   268→  }, [start]);
   269→
   270→  const pause = useCallback(() => {
   271→    const audio = audioRef.current;
   272→    if (audio && !audio.paused) {
   273→      audio.pause();
   274→      setState("paused");
   275→    }
   276→  }, []);
   277→
   278→  const resume = useCallback(() => {
   279→    const audio = audioRef.current;
   280→    if (audio && audio.paused && state !== "needsTap") {
   281→      void audio.play().then(() => setState("playing")).catch(() => {});
   282→    }
   283→  }, [state]);
   284→
   285→  return { state, toggle, pause, resume };
   286→}

   370→function VideoBlockView({ block, index }: { block: BlockDoc; index: number }) {

   372→  const videoRef = useRef<HTMLVideoElement | null>(null);

   381→      v.pause();
   382→      setPlaying(false);

   387→    <motion.div
   388→      initial={{ opacity: 0, y: 22, scale: 0.97 }}

   486→        // ignore

   516→      aria-label={playing ? `Pause ${song.title}` : `Play ${song.title}`}

   683→function CountdownBlockView({
   684→  block,
   685→  index,
   686→  done,
   687→  onDone,
   688→}: {
   689→  block: BlockDoc;
   690→  index: number;
   691→  done: boolean;
   692→  onDone: () => void;
   693→}) {
   694→  const minutes = block.data?.minutes ?? 1;
   695→  const totalMs = minutes * 60_000;
   696→  // Accelerated preview: any countdown resolves within ~4.5s.
   697→  const [elapsed, setElapsed] = useState(0);
   698→  const [running, setRunning] = useState(true);
   699→
   700→  useEffect(() => {
   701→    if (done || !running) return;
   702→    const startedAt = Date.now();
   703→    const id = window.setInterval(() => {
   704→      const t = (Date.now() - startedAt) / 4500; // 0 → 1 over 4.5s
   705→      if (t >= 1) {
   706→        setElapsed(1);
   707→        setRunning(false);
   708→        onDone();
   709→      } else {
   710→        setElapsed(t);
   711→      }
   712→    }, 60);
   713→    return () => window.clearInterval(id);
   714→  }, [done, running, onDone]);
   715→
   716→  const remainingMs = Math.max(0, totalMs * (1 - elapsed));
   717→  const mm = Math.floor(remainingMs / 60000);
   718→  const ss = Math.floor((remainingMs % 60000) / 1000);
   719→  const pct = Math.round(elapsed * 100);
   720→
   721→  return (
   722→    <motion.div
   723→      initial={{ opacity: 0, y: 18 }}
   724→      animate={{ opacity: 1, y: 0 }}
   725→      transition={{ delay: 0.12 + index * 0.09, duration: 0.4, ease: "easeOut" }}
   726→      className="flex w-full flex-col items-center"
   727→    >
   728→      <p className="text-[11px] font-bold uppercase tracking-[0.2em] text-white/60">Unlocks in</p>
   729→      <p
   730→        className={cn(
   731→          "mt-2 font-extrabold tabular-nums tracking-tight transition-colors",
   732→          done ? "text-[34px] text-[#30D158] md:text-[42px]" : "text-[44px] text-white md:text-[56px]"
   733→        )}
   734→      >
   735→        {done ? "Open" : `${String(mm).padStart(2, "0")} : ${String(ss).padStart(2, "0")}`}
   736→      </p>
   737→      <div className="mt-4 h-[6px] w-full max-w-[280px] overflow-hidden rounded-full bg-white/15">
   738→        <div
   739→          className={cn("h-full rounded-full transition-[width] duration-75", done ? "bg-[#30D158]" : "bg-[#FF9F0A]")}
   740→          style={{ width: `${done ? 100 : pct}%` }}
   741→        />
   742→      </div>
   743→      <p className="mt-3 flex items-center gap-1.5 text-[12.5px] font-medium text-white/60">
   744→        <Clock size={12} aria-hidden />
   745→        {done ? "The wait is over" : "Time moves fast in this preview"}
   746→      </p>
   747→      {!done ? (
   748→        <button
   749→          type="button"
   750→          onClick={(e) => {
   751→            e.stopPropagation();
   752→            setRunning(false);
   753→            setElapsed(1);
   754→            onDone();
   755→          }}
   756→          className="mt-4 rounded-full bg-white/12 px-4 py-2 text-[12.5px] font-semibold text-white backdrop-blur-md transition-transform active:scale-95"
   757→        >
   758→          Skip the wait
   759→        </button>
   760→      ) : null}
   761→    </motion.div>
