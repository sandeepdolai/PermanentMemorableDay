   106→/** Pretty countdown label from a minutes value */
   107→function countdownLabel(minutes?: number): string | null {
   108→  if (!minutes) return null;
   109→  if (minutes < 60) return `${minutes} min`;
   110→  if (minutes < 1440) {
   111→    const h = minutes / 60;
   112→    return `${h} hour${h > 1 ? "s" : ""}`;
   113→  }
   114→  const d = minutes / 1440;
   115→  return `${d} day${d > 1 ? "s" : ""}`;
   116→}

   136→function seedScenes(opts: BuilderOptions): Scene[] {
   137→  // Edit-in-Builder restore — the full authored document wins over sketching.
   138→  if (opts.doc && opts.doc.scenes.length > 0) {
   139→    return opts.doc.scenes.map((s) => ({ ...s, blocks: s.blocks.map((b) => ({ ...b })) }));
   140→  }
   141→  if (opts.ai) {
   142→    const scenes: Scene[] = [
   143→      { id: "s1", blocks: [{ id: "b1", type: "text" }, { id: "b2", type: "photo" }] },
   144→      { id: "s2", blocks: [{ id: "b3", type: "gift" }, { id: "b4", type: "confetti" }] },
   145→      { id: "s3", blocks: [{ id: "b5", type: "text" }, { id: "b6", type: "cta" }] },
   146→    ];
   147→    if (opts.seedText) {
   148→      scenes[0].blocks = [{ id: "bAi0", type: "text", text: opts.seedText }, ...scenes[0].blocks];
   149→    }
   150→    return scenes;
   151→  }
   152→  const count = Math.max(1, Math.min(opts.scenes ?? 1, MAX_SCENES));
   153→  const scenes: Scene[] = [];
   154→  for (let i = 0; i < count; i++) {
   155→    const pattern = DRAFT_PATTERNS[(i + (opts.cover ?? 0)) % DRAFT_PATTERNS.length];
   156→    scenes.push({
   157→      id: `s${i + 1}`,
   158→      blocks: pattern.map((t, j) => ({ id: `b${i * 4 + j + 1}`, type: t })),
   159→    });
   160→  }
   161→  if (opts.initialBlock && BLOCK_BY_TYPE[opts.initialBlock]) {
   162→    scenes[0].blocks.push({ id: "bSeed", type: opts.initialBlock });
   163→  }
   164→  if (opts.seedText) {
   165→    scenes[0].blocks = [{ id: "bAi0", type: "text", text: opts.seedText }, ...scenes[0].blocks];
   166→  }
   167→  return scenes;
   168→}

   460→
   461→function BlockCard({
   462→  block,
   463→  cover,
   464→  scenePos,
   465→  active,
   466→  onSelect,
   467→  onRemove,
   468→  onEdit,
   469→  onGripDown,
   470→}: {
   471→  block: Block;
   472→  cover: number;
   473→  scenePos: number;
   474→  active: boolean;
   475→  onSelect: () => void;
   476→  onRemove: () => void;
   477→  onEdit: () => void;
   478→  onGripDown: (e: React.PointerEvent) => void;
   479→}) {
   480→  const controls = useDragControls();
   481→  const def = BLOCK_BY_TYPE[block.type];
   482→  return (
   483→    <Reorder.Item
   484→      value={block}
   485→      dragListener={false}
   486→      dragControls={controls}
   487→      initial={{ opacity: 0, y: 14, scale: 0.97 }}
   488→      animate={{ opacity: 1, y: 0, scale: 1 }}
   489→      exit={{ opacity: 0, scale: 0.94, transition: { duration: 0.15 } }}
   490→      transition={{ type: "spring", stiffness: 420, damping: 32 }}
   491→      role="button"
   492→      tabIndex={0}
   493→      aria-pressed={active}
   494→      aria-label={`${def?.label ?? "Block"} in scene ${scenePos}. ${active ? "Selected" : "Tap to select"}. Drag the handle to reorder.`}
   495→      onClick={onSelect}
   496→      onKeyDown={(e) => {

   740→/** Row preview + snippet playback state machine for one shared <audio>. */
   741→function usePreviewAudio() {
   742→  const audioRef = useRef<HTMLAudioElement | null>(null);
   743→  const [playingId, setPlayingId] = useState<string | null>(null);
   744→  const stopRef = useRef<(() => void) | null>(null);
   745→
   746→  const stop = useCallback(() => {
   747→    stopRef.current?.();
   748→    stopRef.current = null;
   749→    if (audioRef.current) {
   750→      audioRef.current.pause();
   751→      audioRef.current = null;
   752→    }
   753→    setPlayingId(null);
   754→  }, []);
   755→
   756→  /** Plays a clip — from `start` for `length` seconds (preview clips are ~30s). */
   757→  const play = useCallback(
   758→    (id: string, url: string, start: number, length: number) => {
   759→      stop();
   760→      const audio = new Audio(url);
   761→      audioRef.current = audio;
   762→      setPlayingId(id);
   763→      const onTime = () => {
   764→        if (audio.currentTime >= start + length) {
   765→          stop();
   766→        }
   767→      };
   768→      const onReady = () => {
   769→        try {
   770→          audio.currentTime = Math.min(start, Math.max(0, (audio.duration || 30) - 0.5));
   771→        } catch {
   772→          // seek before metadata — the assignment still queues on most browsers
   773→        }
   774→        void audio.play().catch(() => stop());
   775→      };
   776→      const onEnd = () => stop();
   777→      audio.addEventListener("loadedmetadata", onReady, { once: true });
   778→      audio.addEventListener("timeupdate", onTime);
   779→      audio.addEventListener("ended", onEnd);
   780→      audio.addEventListener("error", onEnd);
   781→      // If metadata is already available (cached), loadedmetadata may not fire.
   782→      if (audio.readyState >= 1) onReady();
   783→      stopRef.current = () => {
   784→        audio.removeEventListener("timeupdate", onTime);
   785→        audio.removeEventListener("loadedmetadata", onReady);
   786→        audio.removeEventListener("ended", onEnd);
   787→        audio.removeEventListener("error", onEnd);
   788→      };
   789→    },
   790→    [stop]
   791→  );
   792→
   793→  // Stop on unmount (sheet close)
   794→  useEffect(() => () => stop(), [stop]);
   795→
   796→  return { playingId, play, stop };
   797→}
