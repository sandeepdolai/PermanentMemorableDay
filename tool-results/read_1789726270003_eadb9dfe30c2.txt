    15→interface Particle {
    16→  id: number;
    17→  x: number;

    19→  size: number;

    25→/** One-shot confetti burst (gift reveal) */
    26→function ConfettiBurst() {
    27→  const [particles] = useState<Particle[]>(() =>
    28→    Array.from({ length: 28 }, (_, i) => {
    29→      const angle = (i / 28) * 360 + (Math.random() - 0.5) * 18;
    30→      const distance = 90 + Math.random() * 120;
    31→      const rad = (angle * Math.PI) / 180;
    32→      return {
    33→        id: i,
    34→        x: Math.cos(rad) * distance,
    35→        y: Math.sin(rad) * distance,
    36→        size: 6 + Math.random() * 7,
    37→        rotate: Math.random() * 720 - 360,
    38→        delay: Math.random() * 0.14,
    39→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
    40→      };
    41→    })
    42→  );
    43→
    44→  return (
    45→    <div aria-hidden className="pointer-events-none absolute inset-0 flex items-center justify-center">
    46→      {particles.map((p) => (
    47→        <motion.span
    48→          key={p.id}
    49→          initial={{ x: 0, y: 0, opacity: 1, rotate: 0, scale: 0.5 }}
    50→          animate={{ x: p.x, y: p.y, opacity: 0, rotate: p.rotate, scale: 1 }}
    51→          transition={{ duration: 1 + p.delay * 2.4, ease: "easeOut", delay: p.delay }}
    52→          className="absolute rounded-[2px]"
    53→          style={{ width: p.size, height: p.size * 0.62, backgroundColor: p.color }}
    54→        />
    55→      ))}
    56→    </div>
    57→  );
    58→}
