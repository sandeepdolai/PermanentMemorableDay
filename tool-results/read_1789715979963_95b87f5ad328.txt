    81→/** Deterministic draft layouts — abstract, no themed content */
    82→const DRAFT_PATTERNS: string[][] = [
    83→  ["text", "photo"],
    84→  ["video", "text"],

    93→function seedScenes(opts: BuilderOptions): Scene[] {
    94→  if (opts.ai) {
    95→    const scenes: Scene[] = [
    96→      { id: "s1", blocks: [{ id: "b1", type: "text" }, { id: "b2", type: "photo" }] },
    97→      { id: "s2", blocks: [{ id: "b3", type: "gift" }, { id: "b4", type: "confetti" }] },
    98→      { id: "s3", blocks: [{ id: "b5", type: "text" }, { id: "b6", type: "cta" }] },
    99→    ];
   100→    if (opts.seedText) {
   101→      scenes[0].blocks = [{ id: "bAi0", type: "text", text: opts.seedText }, ...scenes[0].blocks];
   102→    }
   103→    return scenes;
   104→  }
   105→  const count = Math.max(1, Math.min(opts.scenes ?? 1, MAX_SCENES));
   106→  const scenes: Scene[] = [];
   107→  for (let i = 0; i < count; i++) {
   108→    const pattern = DRAFT_PATTERNS[(i + (opts.cover ?? 0)) % DRAFT_PATTERNS.length];
   109→    scenes.push({
   110→      id: `s${i + 1}`,
   111→      blocks: pattern.map((t, j) => ({ id: `b${i * 4 + j + 1}`, type: t })),
   112→    });
   113→  }
   114→  if (opts.initialBlock && BLOCK_BY_TYPE[opts.initialBlock]) {
   115→    scenes[0].blocks.push({ id: "bSeed", type: opts.initialBlock });
   116→  }
   117→  if (opts.seedText) {
   118→    scenes[0].blocks = [{ id: "bAi0", type: "text", text: opts.seedText }, ...scenes[0].blocks];
   119→  }
   120→  return scenes;
   121→}

   286→
   287→function BlockCard({
   288→  block,
   289→  cover,
   290→  scenePos,
   291→  active,
   292→  onSelect,
   293→  onRemove,
   294→  onGripDown,
   295→}: {
   296→  block: Block;
   297→  cover: number;
   298→  scenePos: number;
   299→  active: boolean;
   300→  onSelect: () => void;
   301→  onRemove: () => void;
   302→  onGripDown: (e: React.PointerEvent) => void;
   303→}) {
   304→  const controls = useDragControls();
   305→  const def = BLOCK_BY_TYPE[block.type];
   306→  return (
   307→    <Reorder.Item
   308→      value={block}
   309→      dragListener={false}
   310→      dragControls={controls}
   311→      initial={{ opacity: 0, y: 14, scale: 0.97 }}
   312→      animate={{ opacity: 1, y: 0, scale: 1 }}
   313→      exit={{ opacity: 0, scale: 0.94, transition: { duration: 0.15 } }}
   314→      transition={{ type: "spring", stiffness: 420, damping: 32 }}
   315→      role="button"
   316→      tabIndex={0}
   317→      aria-pressed={active}
   318→      aria-label={`${def?.label ?? "Block"} in scene ${scenePos}. ${active ? "Selected" : "Tap to select"}. Drag the handle to reorder.`}
   319→      onClick={onSelect}
   320→      onKeyDown={(e) => {
   321→        if (e.key === "Enter" || e.key === " ") {
   322→          e.preventDefault();
   323→          onSelect();
   324→        }
   325→      }}
   326→      className={cn(
   327→        "relative cursor-pointer rounded-[18px] border bg-white p-3.5 outline-none transition-shadow",
   328→        "focus-visible:ring-2 focus-visible:ring-[#007AFF]",
   329→        active
   330→          ? "border-[#007AFF] shadow-[0_0_0_3px_rgba(0,122,255,0.12),0_12px_28px_-14px_rgba(0,122,255,0.35)]"
   331→          : "border-[#1D1D1F]/[0.07] card-shadow"
   332→      )}
   333→    >
   334→      <div className="flex items-start gap-2.5">
   335→        <button
   336→          type="button"
   337→          aria-label={`Reorder ${def?.label ?? "block"} in scene ${scenePos}`}
   338→          onPointerDown={(e) => {
   339→            onGripDown(e);
   340→            controls.start(e);
   341→          }}
   342→          onClick={(e) => e.stopPropagation()}
   343→          className={cn(
   344→            "mt-0.5 shrink-0 cursor-grab touch-none rounded-md p-0.5 text-[#C7C7CC] transition-colors active:cursor-grabbing hover:text-[#1D1D1F]/50",
   345→            active && "text-[#007AFF]"
   346→          )}
   347→        >
   348→          <GripVertical size={15} strokeWidth={2.2} aria-hidden />
   349→        </button>
   350→        <div className="min-w-0 flex-1">
   351→          <BlockPreview type={block.type} cover={cover} text={block.text} />
   352→        </div>
   353→      </div>
   354→
   355→      {/* Block meta + delete */}
   356→      <div className="mt-3 flex items-center justify-between border-t border-[#1D1D1F]/[0.05] pt-2.5">
   357→        <span className="flex items-center gap-1.5 text-[10.5px] font-bold uppercase tracking-[0.06em] text-[#AAAAAA]">
   358→          {def ? (
   359→            <def.icon size={11} strokeWidth={2.4} aria-hidden style={{ color: def.tint }} />
   360→          ) : null}
   361→          {def?.label ?? block.type}
   362→        </span>
   363→        {active ? (
   364→          <button
   365→            type="button"
   366→            onClick={(e) => {
   367→              e.stopPropagation();
   368→              onRemove();
   369→            }}
   370→            onKeyDown={(e) => e.stopPropagation()}
   371→            aria-label={`Remove ${def?.label ?? "block"}`}
   372→            className="relative z-10 flex items-center gap-1 rounded-full bg-[#FF375F]/[0.1] px-2.5 py-1 text-[11px] font-semibold text-[#FF375F] transition-transform active:scale-90"
   373→          >
   374→            <Trash2 size={11} strokeWidth={2.4} aria-hidden /> Remove
   375→          </button>
   376→        ) : null}
   377→      </div>
   378→    </Reorder.Item>
   379→  );
   380→}

   466→export function ExperienceBuilder({ opts, onClose }: { opts: BuilderOptions; onClose: () => void }) {
   467→  const { notify, openMoment, openShare, sheet, player, aiInsertRef, openComposer } = useMD();
   468→  const [title, setTitle] = useState(opts.title ?? "Untitled Experience");
   469→  const [editing, setEditing] = useState(false);
   470→  const [scenes, setScenes] = useState<Scene[]>(() => seedScenes(opts));
   471→  const [sceneIdx, setSceneIdx] = useState(0);
   472→  const [selectedBlock, setSelectedBlock] = useState<string | null>(null);
   473→  const [reorderMode, setReorderMode] = useState(false);
   474→  const [past, setPast] = useState<Snapshot[]>([]);
   475→  const [future, setFuture] = useState<Snapshot[]>([]);
   476→  const idRef = useRef(100);
   477→  const blocksEndRef = useRef<HTMLDivElement>(null);
   478→  const blockDragStarted = useRef(false);
   479→  const sceneDragStarted = useRef(false);
   480→  const cover = opts.cover ?? 5;
   481→
   482→  const scene = scenes[Math.min(sceneIdx, scenes.length - 1)];
   483→  const scenePos = sceneIdx + 1;
   484→
   485→  /* ---------- Undo / redo ---------- */
   486→
   487→  const snapshot = useCallback(
   488→    (label: string): Snapshot => ({ label, title, scenes }),
   489→    [title, scenes]
   490→  );
   491→
   492→  /** Push the CURRENT state onto the past stack (call before every mutation) */
   493→  const pushHistory = useCallback(
   494→    (label: string) => {
   495→      setPast((p) => [...p.slice(-(HISTORY_CAP - 1)), snapshot(label)]);
   496→      setFuture([]);
   497→    },
   498→    [snapshot]
   499→  );
   500→
   501→  const applySnapshot = useCallback((s: Snapshot) => {
   502→    setTitle(s.title);

   662→    window.addEventListener("keydown", onKey);
   663→    return () => window.removeEventListener("keydown", onKey);
   664→  }, [sheet, player, editing, onClose, undo, redo]);
